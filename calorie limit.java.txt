import java.util.*;
import java.lang.*;
import java.io.*;

class Codechef
{
	public static void main (String[] args) throws java.lang.Exception
	{
		// your code goes here
		Scanner sc=new Scanner(System.in);
		int t=sc.nextInt();
		while(t-->0){
		    int n=sc.nextInt();
		    int k=sc.nextInt();
		    int count=0;
		    int sum=0;
		    boolean stopped=false;
		    for(int i=0;i<n;i++){
		        int calories=sc.nextInt();
		        if(!stopped && sum+calories<=k){
		            sum=sum + calories;
		            count++;
		        }else{
		            stopped=true;
		        }
		    }
		    System.out.println(count);
		}

	}
}
