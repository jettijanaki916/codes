import java.util.*;
import java.lang.*;
import java.io.*;

class Codechef
{
	public static void main (String[] args) throws java.lang.Exception
	{
		// your code goes here
		Scanner sc=new Scanner(System.in);
		int t=sc.nextInt();
		while(t-->0){
		    int n=sc.nextInt();
		    int m=sc.nextInt();
		    if(9*n < m*10){
		        System.out.println("ONLINE");
		    }else if(9*n > m*10){
		        System.out.println("DINING");
		    }else{
		        System.out.println("EITHER");
		    }
		}

	}
}
