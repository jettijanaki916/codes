import java.util.*;
import java.lang.*;
import java.io.*;

class Codechef
{
	public static void main (String[] args) throws java.lang.Exception
	{
		// your code goes here
		Scanner sc=new Scanner(System.in);
		int t=sc.nextInt();
		while(t-->0){
		    int a=sc.nextInt();
		    int b=sc.nextInt();
		    if((100-a) < (200-2*b)){
		        System.out.println("FIRST");
		    }else if((200-2*b) < (100-a)){
		        System.out.println("SECOND");
		    }else {
		        System.out.println("BOTH");
		    }
		}

	}
}
