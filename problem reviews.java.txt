import java.util.*;
import java.lang.*;
import java.io.*;

class Codechef
{
	public static void main (String[] args) throws java.lang.Exception
	{
		// your code goes here
		Scanner sc=new Scanner(System.in);
		int t=sc.nextInt();
		while(t-->0){
		    int n=sc.nextInt();
		    boolean good=true;
		    for(int i=0;i<n;i++){
		        int score=sc.nextInt();
		        if(score<=4){
		            good=false;
		        }
		    }
		    if(good){
		        System.out.println("YES");
		    }else{
		        System.out.println("NO");
		    }
		        

		}
		    
	}

}
