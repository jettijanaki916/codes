import java.util.*;
import java.lang.*;
import java.io.*;

class Codechef
{
	public static void main (String[] args) throws java.lang.Exception
	{
		// your code goes here
		Scanner sc=new Scanner(System.in);
		int t=sc.nextInt();
		while(t-->0){
		    int n=sc.nextInt();
		    int x=sc.nextInt();
		    int p=sc.nextInt();
		    int marks=4*x-n;
		    if(marks>=p){
		        System.out.println("PASS");
		    }else{
		        System.out.println("FAIL");
		    }
		}

	}
}
