import java.util.*;
import java.lang.*;
import java.io.*;

class Codechef
{
	public static void main (String[] args) throws java.lang.Exception
	{
		// your code goes here
		Scanner sc=new Scanner(System.in);
		int t=sc.nextInt();
		while(t-->0){
		    int n=sc.nextInt();
		    for(int i=1;i<=10;i++){
		        if(n==i){
		            System.out.println("Lower Double");
		        }
		    }
		    for(int i=11;i<=15;i++){
		        if(n==i){
		            System.out.println("Lower Single");
		        }
		    }for(int i=16;i<=25;i++){
		        if(n==i){
		            System.out.println("Upper Double");
		        }
		    }
		    for(int i=26;i<=30;i++){
		        if(n==i){
		            System.out.println("Upper Single");
		        }
		    }
		}

	}
}
